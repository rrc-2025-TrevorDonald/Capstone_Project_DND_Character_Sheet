import request from "supertest";
import { Request, Response, NextFunction } from "express";
import app from "../src/app";
import * as charController from "../src/api/v1/controllers/charController";
import { HTTP_STATUS } from "../src/constants/httpConstants";

jest.mock("../src/api/v1/controllers/charController", () => ({
    getChars: jest.fn((_req: Request, res: Response) =>
        res.status(HTTP_STATUS.OK).send()
    ),
    getChar: jest.fn((_req: Request, res: Response) =>
        res.status(HTTP_STATUS.OK).send()
    ),
    getUserChars: jest.fn((_req: Request, res: Response) =>
        res.status(HTTP_STATUS.OK).send()
    ),
    createItem: jest.fn((_req: Request, res: Response) =>
        res.status(HTTP_STATUS.CREATED).send()
    ),
    updateItem: jest.fn((_req: Request, res: Response) =>
        res.status(HTTP_STATUS.OK).send()
    ),
    deleteItem: jest.fn((_req: Request, res: Response) =>
        res.status(HTTP_STATUS.OK).send()
    ),
}));

// Mocking authentication, essentially bypassing the actual authentication process by calling next()
jest.mock("../src/api/v1/middleware/authenticate", () => {
    return jest.fn((_req: Request, _res: Response, next: NextFunction) =>
        next()
    );
});

// Mocking authorization, essentially bypassing the actual authentication process by calling next() and passing empty _options
jest.mock("../src/api/v1/middleware/authorize", () => {
    return jest.fn(
        (_mockOptions) => (_req: Request, _res: Response, next: NextFunction) =>
            next()
    );
});

describe("Char Routes", () => {
    afterEach(() => {
        jest.clearAllMocks();
    });

    describe("POST /api/v1/chars/", () => {
        it("should call createChar controller with valid data", async () => {
            const mockChar = {
                charName: "Test",
                email: "test@test.com",
            };

            await request(app)
                .post("/api/v1/chars")
                .set("Authorization", "Bearer mockedToken")
                .send(mockChar);
            expect(charController.createChar).toHaveBeenCalled();
        });
    });

    describe("GET /api/v1/chars/all", () => {
        it("should call createChar controller with valid data", async () => {
            await request(app)
                .get("/api/v1/chars/")
                .set("Authorization", "Bearer mockedToken")
            expect(charController.getChars).toHaveBeenCalled();
        });
    });

    describe("POST /api/v1/chars/:email", () => {
        it("should call createChar controller with valid data", async () => {
            await request(app)
                .get("/api/v1/chars/test")
                .set("Authorization", "Bearer mockedToken")
            expect(charController.createChar).toHaveBeenCalled();
        });
    });
});
